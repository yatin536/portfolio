# Production AWS Kinesis & Databricks PySpark Pipeline
from pyspark.sql import SparkSession, functions as F

spark = SparkSession.builder \
    .appName("AWSKinesisDatabricksPipeline") \
    .config("spark.sql.shuffle.partitions", 200) \
    .getOrCreate()

# 1. Ingest streaming delta from AWS Kinesis & S3 Data Lake
stream_df = spark.readStream.format("kinesis") \
    .option("streamName", "analytics-event-stream") \
    .option("region", "us-east-1").load()

# 2. Databricks PySpark distributed cleansing & Delta format
curated_df = stream_df \
    .repartition("partition_date") \
    .withColumn("eff_start_date", F.current_timestamp())

# 3. Micro-partition load into Snowflake & Athena catalog
curated_df.write.format("snowflake") \
    .option("sfWarehouse", "ANALYTICS_WH_XL") \
    .option("clustering_key", "(region_id, event_date)") \
    .mode("append").save()

print("[SUCCESS] Real-time stream processed with 0 errors")
