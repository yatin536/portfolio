# GE Vernova Cash Flow Automation • AWS Data Engg
from pyspark.sql import SparkSession, functions as F

spark = SparkSession.builder \
    .appName("GE_Vernova_CashFlow_Engine") \
    .config("spark.sql.shuffle.partitions", 200) \
    .getOrCreate()

# 1. Ingest streaming financial events via AWS Kinesis
stream_df = spark.readStream.format("kinesis") \
    .option("streamName", "ge-vernova-cashflow-stream") \
    .option("region", "us-east-1").load()

# 2. Stage into Amazon S3 Data Lake & PySpark reconciliation
curated_df = stream_df \
    .repartition("cashflow_date") \
    .withColumn("reconciled_at", F.current_timestamp())

# 3. Load into multi-team Snowflake warehouse & query via Athena
curated_df.write.format("snowflake") \
    .option("sfWarehouse", "FINANCE_CASHFLOW_WH") \
    .option("clustering_key", "(entity_id, cashflow_date)") \
    .mode("append").save()

print("[SUCCESS] Real-time cash flow ingested to S3 Lake & Snowflake")
